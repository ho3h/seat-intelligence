// ---- shared trie primitives (depth-L complete binary trie, leaves are numbers, keyed by bits of the key, high bit first)
// lg(x) = number of bits of x (0 for 0): trie depth for keys 0..x
@lg = (?((0 @lg_s) o) o)
@lg_s = (xm o)
  & xm ~ $([+1] x)
  & x ~ $([>>] $(1 y))
  & @lg ~ (y z)
  & z ~ $([+1] o)

// zt(L): trie of depth L with all leaves 0
@zt = (?((0 @zt_node) o) o)
@zt_node = (lm (a b))
  & lm ~ {l1 l2}
  & @zt ~ (l1 a)
  & @zt ~ (l2 b)

// cnt(t, L): sum of the leaves
@cnt = (t (L o))
  & L ~ ?((@cnt_leaf @cnt_node) (t o))
@cnt_leaf = (x x)
@cnt_node = (lm ((a b) o))
  & lm ~ {l1 l2}
  & @cnt ~ (a (l1 x))
  & @cnt ~ (b (l2 y))
  & x ~ $([+] $(y o))

// ins(t, k, L): set leaf k to 1. Control flow depends on k and L only, so it expands before t exists.
@ins = (t (k (L o)))
  & L ~ ?((@ins_leaf @ins_node) (t (k o)))
@ins_leaf = (* (* 1))
@ins_node = (lm (t (k o)))
  & k ~ {k1 k2}
  & lm ~ {l1 l2}
  & k1 ~ $([>>] $(l1 sh))
  & sh ~ $([&] $(1 b))
  & b ~ ?((@ins_L @ins_R) (t (k2 (l2 o))))
@ins_L = ((l r) (k (L (l2 r))))
  & @ins ~ (l (k (L l2)))
@ins_R = (* ((l r) (k (L (l r2)))))
  & @ins ~ (r (k (L r2)))
