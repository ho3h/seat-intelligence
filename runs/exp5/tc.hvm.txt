// t3_triangle_count (GRAPH-SWING, runs/exp5).
// Data structures built while streaming the edge list, all by pre-expanded keyed updates:
//   M: adjacency matrix, trie over u of row tries over word index, leaves 16-bit words; row u = N+(u) = {v > u}.
//   Q: request trie over vertex id; leaf = (in out) end of a DUP multicast chain. Each edge (u<v) pushes two reply
//      wires x1 (for row u) and x2 (for row v): wires are return addresses, so nothing has to be routed back.
//   S: sum trie over u; edge (u, v) adds popcount(row u AND row v) at leaf u.
// fin: zip M with Q (pure wiring: row w -> multicast chain of leaf w), then reduce S. Each triangle a<b<c is counted
// once, by the edge (a, b).
@prog = ((n es) out)
  & n ~ ?((@tc_zero @tc_pos) (es out))
@tc_zero = (* 0)
@tc_pos = (nm1 (es out))
  & nm1 ~ {a b}
  & @lg ~ (a D)
  & b ~ $([>>] $(4 bw))
  & @lg ~ (bw W)
  & D ~ {D1 {D2 {D3 D4}}}
  & W ~ {W1 W2}
  & @zm ~ (D1 (W1 M0))
  & @zq ~ (D2 Q0)
  & @zt ~ (D3 S0)
  & @w_blk ~ (es ((D4 (W2 (M0 (Q0 S0)))) out))

@tc_step = ((D (W (M (Q S)))) ((a b) (Dn (Wn (M2 (Q2 S2))))))
  & a ~ {a1 a2}
  & b ~ {b1 b2}
  & a1 ~ $([<] $(b1 lt))
  & lt ~ ?((@tc_swap @tc_keep) (a2 (b2 (u v))))
  & u ~ {u1 {u2 u3}}
  & v ~ {v1 v2}
  & D ~ {D1 {D2 {D3 {D4 Dn}}}}
  & W ~ {W1 {W2 Wn}}
  & @sb ~ (M (u1 (D1 ((v1 W1) M2))))
  & @pu ~ (Q (u2 (D2 (x1 Qa))))
  & @pu ~ (Qa (v2 (D3 (x2 Q2))))
  & @pa ~ (x1 (x2 (W2 c)))
  & @ad ~ (S (u3 (D4 (c S2))))
@tc_swap = (a (b (b a)))
@tc_keep = (* (a (b (a b))))

@tc_fin = ((D (W (M (Q S)))) o)
  & D ~ {D1 D2}
  & W ~ *
  & @zmq ~ (M (Q D1))
  & @cnt ~ (S (D2 o))

// set bit v in row u: row trie keyed by v>>4, word |= 1 << (v & 15)
@sb_act = (row ((v W) o))
  & v ~ {v1 v2}
  & v1 ~ $([>>] $(4 kw))
  & v2 ~ $([&] $(15 bi))
  & 1 ~ $([<<] $(bi mask))
  & @sr ~ (row (kw (W (mask o))))
@sr_act = (w (mask o))
  & w ~ $([|] $(mask o))
// push a reply wire onto the multicast chain of a leaf
@pu_act = ((i o) (x (i o2)))
  & o ~ {x o2}
// add c to a leaf
@ad_act = (old (c o))
  & old ~ $([+] $(c o))

// zero matrix: depth-L trie of depth-W zero rows
@zm = (L (W o))
  & L ~ ?((@zm_leaf @zm_node) (W o))
@zm_leaf = (W o)
  & @zt ~ (W o)
@zm_node = (lm (W (a b)))
  & lm ~ {l1 l2}
  & W ~ {w1 w2}
  & @zm ~ (l1 (w1 a))
  & @zm ~ (l2 (w2 b))
// empty request trie: leaves are wire loops (y y)
@zq = (?(((y y) @zq_node) o) o)
@zq_node = (lm (a b))
  & lm ~ {l1 l2}
  & @zq ~ (l1 a)
  & @zq ~ (l2 b)
// deliver row w of M to the multicast chain at leaf w of Q, erase the chain's end
@zmq = (m (q L))
  & L ~ ?((@zmq_leaf @zmq_node) (m q))
@zmq_leaf = (R (R *))
@zmq_node = (lm ((m1 m2) (q1 q2)))
  & lm ~ {l1 l2}
  & @zmq ~ (m1 (q1 l1))
  & @zmq ~ (m2 (q2 l2))
// pa(a, b, W): popcount of the AND of two depth-W word tries
@pa = (a (b (L o)))
  & L ~ ?((@pa_leaf @pa_node) (a (b o)))
@pa_leaf = (a (b o))
  & a ~ $([&] $(b w))
  & @pop16 ~ (w o)
@pa_node = (lm ((a1 a2) ((b1 b2) o)))
  & lm ~ {l1 l2}
  & @pa ~ (a1 (b1 (l1 x)))
  & @pa ~ (a2 (b2 (l2 y)))
  & x ~ $([+] $(y o))
@pop16 = (x o)
  & x ~ {x1 x2}
  & x1 ~ $([>>] $(1 y))
  & y ~ $([&] $(21845 y2))
  & x2 ~ $([-] $(y2 a))
  & a ~ {a1 a2}
  & a1 ~ $([&] $(13107 p))
  & a2 ~ $([>>] $(2 q))
  & q ~ $([&] $(13107 q2))
  & p ~ $([+] $(q2 b))
  & b ~ {b1 b2}
  & b2 ~ $([>>] $(4 r))
  & b1 ~ $([+] $(r c))
  & c ~ $([&] $(3855 c2))
  & c2 ~ {c3 c4}
  & c4 ~ $([>>] $(8 s))
  & c3 ~ $([+] $(s e))
  & e ~ $([&] $(31 o))
