// ---- shared SSSP core (GRAPH-SWING, runs/exp5): frontier Bellman-Ford over tries.
// G: adjacency trie keyed by vertex, leaves are lists of (v w). Dt: distance trie. C: candidate trie of this round.
// One round = one pre-expanded traversal of (G, Dt, C): at each leaf, d' = min(d, c) (accepted only if c <= budget);
// a leaf that improved emits d'+w into the NEXT round's candidate trie for each of its out-edges (min-updates, keyed
// by v, threaded through the traversal in order); flags are OR-reduced; the loop continues while anything improved.
// E = (L b): trie depth and budget (b = 16777215 means no budget).
@loop = ((g (dt (c E))) o)
  & E ~ {E1 {E2 E3}}
  & E1 ~ (L1 *)
  & @zi ~ (L1 ci)
  & E2 ~ {Ea Eb}
  & Ea ~ (Lr *)
  & @rd ~ ((Lr Eb) ((g (dt c)) ((ci co) (g2 (dt2 any)))))
  & @lp ~ ((g2 (dt2 (co (E3 any)))) o)
// lp(state of round k, any_k): round k+1 is expanded now (one round of lookahead); any_k decides whether round k+2
// is created. When nothing improved in round k, round k+1 runs without candidates and its distances are final.
@lp = ((g (dt (c (E any)))) o)
  & E ~ {E1 {E2 E3}}
  & E1 ~ (L1 *)
  & @zi ~ (L1 ci)
  & E2 ~ {Ea Eb}
  & Ea ~ (Lr *)
  & @rd ~ ((Lr Eb) ((g (dt c)) ((ci co) (g2 (dt2 any2)))))
  & any ~ ?((@lp_done @lp_again) ((g2 (dt2 (co (E3 any2)))) o))
@lp_done = ((* (dt (* (* *)))) dt)
@lp_again = (* (st o))
  & @lp ~ (st o)

@rd = ((L E) ((g (dt c)) ((ci co) (g2 (dt2 any)))))
  & L ~ ?((@rd_leaf @rd_node) (E ((g (dt c)) ((ci co) (g2 (dt2 any))))))
@rd_node = (lm (E (((g1 gr) ((d1 dr) (c1 cr))) ((ci co) ((h1 hr) ((e1 er) any))))))
  & lm ~ {l1 l2}
  & E ~ {Ea Eb}
  & @rd ~ ((l1 Ea) ((g1 (d1 c1)) ((ci cm) (h1 (e1 a1)))))
  & @rd ~ ((l2 Eb) ((gr (dr cr)) ((cm co) (hr (er a2)))))
  & a1 ~ $([|] $(a2 any))
@rd_leaf = (E ((g (d c)) ((ci co) (g2 (d2 any)))))
  & c ~ {c1 {c2 c3}}
  & d ~ {dA dB}
  & E ~ {Ex Ey}
  & Ey ~ (* b)
  & c1 ~ $([<] $(dA lt))
  & c2 ~ $([>] $(b gt))
  & lt ~ $([>] $(gt f))
  & f ~ {f1 {any f3}}
  & f1 ~ ?((@rl_keep @rl_take) (dB (c3 nd)))
  & nd ~ {nd1 d2}
  & f3 ~ ?((@rl_quiet @rl_emit) (nd1 ne))
  & Ex ~ (L *)
  & @em ~ (g (ne (ci (g2 (co L)))))
@rl_keep = (d (* d))
@rl_take = (* (* (c c)))
// emission is not gated: every leaf walks its list every round (so the keyed updates expand a round ahead);
// a leaf that did not improve emits the harmless 16776215 (+w), which the budget clamp always rejects

// emit nd+w into the candidate trie for every (v w) of the adjacency list, rebuilding the list
@em = ((?((@em_nil @em_cons) (p (nd (ci (g2 (co L)))))) p) (nd (ci (g2 (co L)))))
@em_nil = (* (* (ci ((0 *) (ci *)))))
@em_cons = (* (((v w) rest) (nd (ci ((1 ((v2 w2) rest2)) (co L))))))
  & v ~ {v1 v2}
  & w ~ {w1 w2}
  & nd ~ {n1 n2}
  & L ~ {La Lb}
  & n1 ~ $([+] $(w1 cand))
  & @mu ~ (ci (v1 (La (cand cm))))
  & @em ~ (rest (n2 (cm (rest2 (co Lb)))))

  & k1 ~ $([<] $(o1 lt))
  & lt ~ ?((@mu_keep @mu_take) (o2 (k2 o)))

// tries with constant leaves: 16777215, and empty lists
@zi = (?((16777215 @zi_node) o) o)
@zi_node = (lm (a b))
  & lm ~ {l1 l2}
  & @zi ~ (l1 a)
  & @zi ~ (l2 b)
@zl = (?(((0 *) @zl_node) o) o)
@zl_node = (lm (a b))
  & lm ~ {l1 l2}
  & @zl ~ (l1 a)
  & @zl ~ (l2 b)

// initial state: (G, Dt = all INF, C0 = 0 at s if s < n), then the loop; o receives the final distance trie
@sssp_start = (n (s (L (b (g o)))))
  & s ~ {s1 s2}
  & n ~ {n1 n2}
  & n1 ~ *
  & s1 ~ $([<] $(n2 inr))
  & inr ~ {i1 i2}
  & s2 ~ $([*] $(i1 key))
  & i2 ~ ?((16777215 (* 0)) val)

  & b1 ~ $([<] $(16776215 small))

  & @loop ~ ((g (D0 (C0 (L4 bc)))) o)
