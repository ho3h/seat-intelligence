// t3_clique_number (GRAPH-SWING-2, runs/exp9): parallel forward-clique enumeration. Rows N+(v) = {u ~ v : u > v} are
// bitsets (word tries). Each clique is enumerated once in increasing order by the recursion
//   clq(E) = max over entries (w, S_w) of E of 1 + clq([(x, S_x & S_w) for (x, S_x) in E if x in S_w]),  clq([]) = 0
// where E holds the candidate vertices with their forward rows restricted to the candidate set. All branches run in
// parallel (each branch copies its small entry list). The top-level lists come from the multicast pattern: every
// edge (lo, hi) pushes a reply wire onto hi's chain and onto lo's entry list, and fin sends (hi, N+(hi)) down each
// chain. Answer = max over v of 1 + clq(E_v); n = 0 gives 0.
@prog = ((n es) out)
  & n ~ ?((@cq_zero @cq_pos) (es out))
@cq_zero = (* 0)
@cq_pos = (nm1 (es out))
  & nm1 ~ {b1 b2}
  & @lg ~ (b1 L)
  & b2 ~ $([>>] $(4 bw))
  & @lg ~ (bw W)
  & L ~ {L1 {L2 {L3 {L4 L5}}}}
  & W ~ {W1 {W2 W3}}
  & @zm ~ (L1 (W1 M0))
  & @zq ~ (L2 Q0)
  & @zl ~ (L3 R0)
  & @w_blk ~ (es (((L4 W2) (M0 (Q0 R0))) (M (Q R))))
  & @zk ~ (M (Q (R (L5 (W3 (0 out))))))
@cq_step = (((L W) (m (q r))) ((a b) ((L4 W2) (m2 (q2 r2)))))
  & a ~ {a1 a2}
  & b ~ {b1 b2}
  & a1 ~ $([<] $(b1 lt))
  & lt ~ ?((@cq_swap @cq_keep) (a2 (b2 (u v))))
  & u ~ {u1 u2}
  & v ~ {v1 v2}
  & L ~ {L1 {L2 {L3 L4}}}
  & W ~ {W1 W2}
  & @sb ~ (m (u1 (L1 ((v1 W1) m2))))
  & @pu ~ (q (v2 (L2 (x q2))))
  & @pn ~ (r (u2 (L3 (x r2))))
@cq_swap = (a (b (b a)))
@cq_keep = (* (a (b (a b))))
@cq_fin = ((* s) s)
// zk(M, Q, R, L, W, base) -> max over leaves v of 1 + clq(E_v)
@zk = (m (q (r (L (W (base o))))))
  & L ~ ?((@zk_leaf @zk_node) (m (q (r (W (base o))))))
@zk_leaf = (nv (((v n1) *) (r (W (v o)))))
  & nv ~ {n1 n2}
  & W ~ {W1 W2}
  & @mapand ~ (r (n2 (W1 e)))
  & @clq ~ (e (W2 c))
  & c ~ $([+1] o)
@zk_node = (lm ((ma mb) ((qa qb) ((ra rb) (W (base o))))))
  & lm ~ {l1 {l2 l3}}
  & W ~ {W1 W2}
  & base ~ {b1 b2}
  & 1 ~ $([<<] $(l3 half))
  & b2 ~ $([+] $(half bR))
  & @zk ~ (ma (qa (ra (l1 (W1 (b1 x))))))
  & @zk ~ (mb (qb (rb (l2 (W2 (bR y))))))
  & @max ~ (x (y o))
// mapand(list of (x S), N, W) -> list of (x (S & N))
@mapand = ((?((@ma_nil @ma_cons) (p (N (W o)))) p) (N (W o)))
@ma_nil = (* (* (* (0 *))))
@ma_cons = (* (((x s) rest) (N (W (1 ((x s2) r2))))))
  & N ~ {N1 N2}
  & W ~ {W1 W2}
  & @wand ~ (W1 (s (N1 s2)))
  & @mapand ~ (rest (N2 (W2 r2)))
// clq(E, W)
@clq = (e (W o))
  & e ~ {e1 e2}
  & @cqi ~ (e1 (e2 (W o)))
@cqi = ((?((@cqi_nil @cqi_cons) (p (f (W o)))) p) (f (W o)))
@cqi_nil = (* (* (* 0)))
@cqi_cons = (* (((* s) rest) (f (W o))))
  & f ~ {f1 f2}
  & W ~ {W1 {W2 W3}}
  & @flt ~ (f1 (s (W1 ch)))
  & @clq ~ (ch (W2 c))
  & c ~ $([+1] a)
  & @cqi ~ (rest (f2 (W3 b)))
  & @max ~ (a (b o))
// flt(E, S, W) -> [(x, S_x & S) for (x, S_x) in E if x in S]
@flt = ((?((@flt_nil @flt_cons) (p (s (W o)))) p) (s (W o)))
@flt_nil = (* (* (* (0 *))))
@flt_cons = (* (((x sx) rest) (s (W o))))
  & s ~ {s1 {s2 s3}}
  & W ~ {W1 {W2 W3}}
  & x ~ {x1 x2}
  & @bt ~ (s1 (x1 (W1 in)))
  & in ~ ?((@flt_skip @flt_keep) (x2 (sx (s2 (W2 (r o))))))
  & @flt ~ (rest (s3 (W3 r)))
@flt_skip = (* (* (* (* (r r)))))
@flt_keep = (* (x (sx (s (W (r (1 ((x sa) r))))))))
  & @wand ~ (W (sx (s sa)))
// bt(S, x, W): bit x of word-trie set S
@bt = (s (x (W o)))
  & x ~ {x1 x2}
  & x1 ~ $([>>] $(4 j))
  & x2 ~ $([&] $(15 b))
  & @lk ~ (s (j (W w)))
  & w ~ $([>>] $(b w2))
  & w2 ~ $([&] $(1 o))
@wand = (W (a (b o)))
  & W ~ ?((@wand_leaf @wand_node) (a (b o)))
@wand_leaf = (a (b o))
  & a ~ $([&] $(b o))
@wand_node = (wm ((a1 a2) ((b1 b2) (x y))))
  & wm ~ {w1 w2}
  & @wand ~ (w1 (a1 (b1 x)))
  & @wand ~ (w2 (a2 (b2 y)))
